-- Multi-Class Vocabulary LMS with Strict Class-Level Data Isolation
-- ==============================================================================

-- 1. Create 'classes' table
CREATE TABLE IF NOT EXISTS classes (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name TEXT NOT NULL,
  class_code TEXT UNIQUE NOT NULL DEFAULT substr(md5(random()::text), 0, 8),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Create 'profiles' table (linked to Supabase Auth)
CREATE TABLE IF NOT EXISTS profiles (
  id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
  role TEXT CHECK (role IN ('teacher', 'student')) DEFAULT 'student',
  full_name TEXT NOT NULL,
  class_id BIGINT REFERENCES classes(id) ON DELETE SET NULL,
  email TEXT,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Create 'lessons' table (Strictly tied to class_id)
CREATE TABLE IF NOT EXISTS lessons (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  class_id BIGINT REFERENCES classes(id) ON DELETE CASCADE NOT NULL,
  title TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Create 'vocabulary' table (Strictly tied to class_id and lesson_id)
CREATE TABLE IF NOT EXISTS vocabulary (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  class_id BIGINT REFERENCES classes(id) ON DELETE CASCADE NOT NULL,
  lesson_id BIGINT REFERENCES lessons(id) ON DELETE CASCADE NOT NULL,
  word TEXT NOT NULL,
  meaning TEXT NOT NULL,
  ipa TEXT,
  example TEXT,
  is_grammar BOOLEAN DEFAULT FALSE,
  created_by UUID REFERENCES auth.users(id),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 5. Create 'test_sessions' table (Scoped to user_id and class_id)
CREATE TABLE IF NOT EXISTS test_sessions (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  class_id BIGINT REFERENCES classes(id) ON DELETE CASCADE NOT NULL,
  session_type TEXT NOT NULL, -- 'lesson_based' or 'random'
  test_scope JSONB, -- { "lesson_id": 1 } or "all" (all vocabulary of the class)
  total_questions INTEGER DEFAULT 0,
  correct_count INTEGER DEFAULT 0,
  wrong_count INTEGER DEFAULT 0,
  skipped_count INTEGER DEFAULT 0,
  score_percentage FLOAT DEFAULT 0,
  duration_seconds INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 6. Create 'session_details' table
CREATE TABLE IF NOT EXISTS session_details (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  session_id BIGINT REFERENCES test_sessions(id) ON DELETE CASCADE NOT NULL,
  word_id BIGINT REFERENCES vocabulary(id) ON DELETE SET NULL,
  user_answer TEXT,
  correct_answer TEXT,
  is_correct BOOLEAN,
  is_skipped BOOLEAN DEFAULT FALSE
);

-- ==============================================================================
-- ROW LEVEL SECURITY (RLS) POLICIES - STRICT CLASS ISOLATION
-- ==============================================================================

ALTER TABLE classes ENABLE ROW LEVEL SECURITY;
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE lessons ENABLE ROW LEVEL SECURITY;
ALTER TABLE vocabulary ENABLE ROW LEVEL SECURITY;
ALTER TABLE test_sessions ENABLE ROW LEVEL SECURITY;
ALTER TABLE session_details ENABLE ROW LEVEL SECURITY;

-- Helper function: Get user's role
CREATE OR REPLACE FUNCTION get_user_role() RETURNS TEXT AS $$
  SELECT role FROM profiles WHERE id = auth.uid();
$$ LANGUAGE sql SECURITY DEFINER;

-- Helper function: Get user's active class_id
CREATE OR REPLACE FUNCTION get_user_class_id() RETURNS BIGINT AS $$
  SELECT class_id FROM profiles WHERE id = auth.uid();
$$ LANGUAGE sql SECURITY DEFINER;

-- --- CLASSES POLICIES ---
DROP POLICY IF EXISTS "Public read classes by code or member" ON classes;
CREATE POLICY "Public read classes by code or member" ON classes 
  FOR SELECT USING (true);

DROP POLICY IF EXISTS "Teachers can manage classes" ON classes;
CREATE POLICY "Teachers can manage classes" ON classes 
  FOR ALL USING (get_user_role() = 'teacher');

-- --- PROFILES POLICIES ---
DROP POLICY IF EXISTS "Users can view and update their own profile" ON profiles;
CREATE POLICY "Users can view and update their own profile" ON profiles 
  FOR ALL USING (auth.uid() = id);

DROP POLICY IF EXISTS "Teachers can view student profiles" ON profiles;
CREATE POLICY "Teachers can view student profiles" ON profiles 
  FOR SELECT USING (get_user_role() = 'teacher');

-- --- LESSONS POLICIES (Class Isolated) ---
DROP POLICY IF EXISTS "Students can view lessons of their own class" ON lessons;
CREATE POLICY "Students can view lessons of their own class" ON lessons 
  FOR SELECT USING (
    get_user_role() = 'teacher' OR class_id = get_user_class_id()
  );

DROP POLICY IF EXISTS "Teachers and students can create lessons in their class" ON lessons;
CREATE POLICY "Teachers and students can create lessons in their class" ON lessons 
  FOR INSERT WITH CHECK (
    get_user_role() = 'teacher' OR class_id = get_user_class_id()
  );

DROP POLICY IF EXISTS "Teachers and students can update lessons in their class" ON lessons;
CREATE POLICY "Teachers and students can update lessons in their class" ON lessons 
  FOR UPDATE USING (
    get_user_role() = 'teacher' OR class_id = get_user_class_id()
  );

DROP POLICY IF EXISTS "Teachers can manage lessons" ON lessons;
CREATE POLICY "Teachers can manage lessons" ON lessons 
  FOR ALL USING (get_user_role() = 'teacher');

-- --- VOCABULARY POLICIES (Class Isolated) ---
DROP POLICY IF EXISTS "Students can view vocabulary of their own class" ON vocabulary;
CREATE POLICY "Students can view vocabulary of their own class" ON vocabulary 
  FOR SELECT USING (
    get_user_role() = 'teacher' OR class_id = get_user_class_id()
  );

DROP POLICY IF EXISTS "Teachers and students can add vocabulary to their class" ON vocabulary;
CREATE POLICY "Teachers and students can add vocabulary to their class" ON vocabulary 
  FOR INSERT WITH CHECK (
    get_user_role() = 'teacher' OR class_id = get_user_class_id()
  );

DROP POLICY IF EXISTS "Teachers and students can update vocabulary of their class" ON vocabulary;
CREATE POLICY "Teachers and students can update vocabulary of their class" ON vocabulary 
  FOR UPDATE USING (
    get_user_role() = 'teacher' OR class_id = get_user_class_id()
  );

DROP POLICY IF EXISTS "Teachers and students can delete vocabulary in their class" ON vocabulary;
CREATE POLICY "Teachers and students can delete vocabulary in their class" ON vocabulary 
  FOR DELETE USING (
    get_user_role() = 'teacher' OR class_id = get_user_class_id()
  );

DROP POLICY IF EXISTS "Teachers can manage vocabulary" ON vocabulary;
CREATE POLICY "Teachers can manage vocabulary" ON vocabulary 
  FOR ALL USING (get_user_role() = 'teacher');

-- --- TEST SESSIONS POLICIES ---
DROP POLICY IF EXISTS "Students can create and view their own test sessions" ON test_sessions;
CREATE POLICY "Students can create and view their own test sessions" ON test_sessions 
  FOR ALL USING (
    auth.uid() = user_id OR get_user_role() = 'teacher'
  );

DROP POLICY IF EXISTS "Students can create and view test details" ON session_details;
CREATE POLICY "Students can create and view test details" ON session_details 
  FOR ALL USING (
    EXISTS (
      SELECT 1 FROM test_sessions 
      WHERE test_sessions.id = session_details.session_id 
      AND (test_sessions.user_id = auth.uid() OR get_user_role() = 'teacher')
    )
  );

-- ==============================================================================
-- 7. STUDY SESSIONS TABLE (Ghi nhận toàn bộ thời gian học sinh vào học, kể cả Flashcard)
-- ==============================================================================
CREATE TABLE IF NOT EXISTS study_sessions (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id TEXT NOT NULL,
  user_name TEXT DEFAULT 'Học sinh',
  class_id BIGINT REFERENCES classes(id) ON DELETE CASCADE NOT NULL,
  lesson_id BIGINT REFERENCES lessons(id) ON DELETE SET NULL,
  activity_type TEXT NOT NULL DEFAULT 'flashcard', -- 'flashcard' | 'quiz' | 'vocabulary'
  duration_seconds INTEGER DEFAULT 0,
  cards_viewed INTEGER DEFAULT 0,
  cards_mastered INTEGER DEFAULT 0,
  score FLOAT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE study_sessions ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Public access study_sessions" ON study_sessions;
CREATE POLICY "Public access study_sessions" ON study_sessions FOR ALL USING (true);

-- ==============================================================================
-- 8. MULTI-DEVICE CLOUD ACCOUNTS TABLE (Cross-device Account Synchronization)
-- ==============================================================================
CREATE TABLE IF NOT EXISTS app_users (
  id TEXT PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  password TEXT NOT NULL,
  alt_password TEXT,
  full_name TEXT NOT NULL,
  role TEXT NOT NULL DEFAULT 'student',
  class_id BIGINT REFERENCES classes(id) ON DELETE SET NULL,
  email TEXT,
  streak INTEGER DEFAULT 0,
  completed_lessons INTEGER DEFAULT 0,
  avg_score FLOAT DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE app_users ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Public read app_users" ON app_users;
CREATE POLICY "Public read app_users" ON app_users 
  FOR SELECT USING (true);

DROP POLICY IF EXISTS "Allow modify app_users" ON app_users;
CREATE POLICY "Allow modify app_users" ON app_users 
  FOR ALL USING (true);
